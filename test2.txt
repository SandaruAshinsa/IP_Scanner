import ipaddress
import asyncio
import socket
import http.client
import threading
import tkinter as tk
from tkinter import ttk
from datetime import datetime

STOP_SCAN = False


# ================= UTIL =================

def get_ips(cidr):
    return [str(ip) for ip in ipaddress.ip_network(cidr, strict=False)]


def resolve_ip(host):
    try:
        return socket.gethostbyname(host)
    except:
        return None


def ip_to_cidr(ip):
    return ".".join(ip.split(".")[:3]) + ".0/24"


# ================= SERVICE DETECTION =================

def detect_service(server, banner):
    text = (server + " " + banner).lower()

    if "apache" in text:
        return "Apache"
    elif "nginx" in text:
        return "Nginx"
    elif "iis" in text:
        return "IIS"
    elif "lighttpd" in text:
        return "Lighttpd"
    else:
        return "Unknown"


# ================= HTTP =================

def http_probe(ip, port, timeout):
    try:
        conn = http.client.HTTPConnection(ip, port, timeout=timeout)
        conn.request("GET", "/")
        res = conn.getresponse()

        headers = dict(res.getheaders())
        banner = f"{res.status} {res.reason}"
        server = headers.get("Server", "")

        conn.close()
        return banner, server

    except:
        return "", ""


# ================= SCAN =================

async def scan_port(ip, port, timeout):
    loop = asyncio.get_event_loop()

    def run():
        banner, server = "", ""

        if port in (80, 443):
            banner, server = http_probe(ip, port, timeout)

        return (port, bool(banner), banner, server)

    return await loop.run_in_executor(None, run)


async def scan_ip(ip, ports, timeout, sem):
    global STOP_SCAN

    if STOP_SCAN:
        return None

    async with sem:
        tasks = [scan_port(ip, p, timeout) for p in ports]
        ports_data = await asyncio.gather(*tasks)

        return {"ip": ip, "ports": ports_data}


async def run_scan(cidr, ports, timeout, callback, concurrency):
    global STOP_SCAN
    STOP_SCAN = False

    ips = get_ips(cidr)
    sem = asyncio.Semaphore(concurrency)

    tasks = [scan_ip(ip, ports, timeout, sem) for ip in ips]

    total = len(tasks)
    done = 0

    for f in asyncio.as_completed(tasks):
        if STOP_SCAN:
            break

        res = await f
        done += 1

        if res:
            callback(res, done, total)

    def apply_theme(self):
        if self.dark_mode:
            bg = self.bg_dark
            panel = self.panel_dark
            entry = self.entry_dark
            fg = self.fg_dark
        else:
            bg = "#f5f5f5"
            panel = "#ffffff"
            entry = "#ffffff"
            fg = "#000000"

        self.root.configure(bg=bg)

        self.style.configure("TFrame", background=panel)
        self.style.configure("TLabel", background=panel, foreground=fg)
        self.style.configure("TCheckbutton", background=panel, foreground=fg)
        self.style.configure("TButton", background=entry, foreground=fg)
        self.style.configure("TEntry", fieldbackground=entry, foreground=fg)
        self.style.configure("TNotebook", background=panel)
        self.style.configure("TNotebook.Tab", background=entry, foreground=fg, padding=[10, 5])
        self.style.configure("TProgressbar", background=self.accent, troughcolor=entry)

        self.style.configure("Treeview",
            background=entry,
            foreground=fg,
            fieldbackground=entry,
            rowheight=25
        )

        self.style.configure("Treeview.Heading",
            background=panel,
            foreground=fg
        )
        
        # Text widget colors
        if hasattr(self, 'log_text'):
            if self.dark_mode:
                self.log_text.configure(bg=self.entry_dark, fg=self.fg_dark)
            else:
                self.log_text.configure(bg="white", fg="black")

    def toggle_theme(self):
        self.dark_mode = not self.dark_mode
        self.apply_theme()

    # ================= LOGGING =================

    def add_log(self, message, level="info"):
        timestamp = datetime.now().strftime("%Y/%m/%d %H:%M:%S")
        
        colors = {
            "info": "#4a9eff",
            "warning": "#ff9800",
            "error": "#f44336",
            "success": "#4caf50"
        }
        
        color = colors.get(level, "#e0e0e0")
        
        self.log_text.insert("end", f"{timestamp} ", "timestamp")
        self.log_text.insert("end", f"[{level.upper()}] ", level)
        self.log_text.insert("end", f"{message}\n")
        
        self.log_text.see("end")
        
        # Tags
        self.log_text.tag_configure("timestamp", foreground="#888888")
        self.log_text.tag_configure(level, foreground=color)

    # ================= CONTROL =================

    def start(self):
        self.add_log("Starting profile [Scanner] Network Scan", "info")
        threading.Thread(target=self.run_async, daemon=True).start()

    def run_async(self):
        cidr = self.cidr.get()
        ports = list(map(int, self.ports.get().split(",")))
        timeout = float(self.timeout.get())
        concurrency = int(self.speed.get())
        
        self.add_log(f"Scanning {cidr} on ports {ports}", "info")

        asyncio.run(run_scan(cidr, ports, timeout, self.update_ui, concurrency))
        
        self.add_log("Scan completed", "success")

    def stop(self):
        global STOP_SCAN
        STOP_SCAN = True
        self.add_log("Stopping scan...", "warning")

    def clear(self):
        self.tree.delete(*self.tree.get_children())
        self.progress["value"] = 0
        self.log_text.delete("1.0", "end")
        self.add_log("Cleared", "info")

    # ================= FILTER =================

    def filter_results(self, event=None):
        keyword = self.search.get().lower()

        for item in self.tree.get_children():
            values = self.tree.item(item)["values"]
            text = " ".join(map(str, values)).lower()

            if keyword in text:
                self.tree.reattach(item, "", "end")
            else:
                self.tree.detach(item)

    # ================= AUTO CIDR =================

    def auto_cidr(self, event=None):
        ip = resolve_ip(self.host.get())
        if ip:
            self.cidr.delete(0, tk.END)
            self.cidr.insert(0, ip_to_cidr(ip))
            self.add_log(f"Auto-detected CIDR: {ip_to_cidr(ip)}", "info")

    # ================= UPDATE =================

    def update_ui(self, res, done, total):
        self.progress["maximum"] = total
        self.progress["value"] = done

        for port, open_, banner, server in res["ports"]:

            if self.only_open.get() and not open_:
                continue

            service = detect_service(server, banner)

            status = "OPEN" if open_ else "CLOSED"
            tag = "open" if open_ else "closed"
            
            # Format test result
            if open_:
                test_result = f"Latency: {banner} / Service: {service}"
            else:
                test_result = "Connection Refused"

            # Determine type based on port
            if port == 80:
                proto_type = "HTTP"
            elif port == 443:
                proto_type = "HTTPS"
            else:
                proto_type = "TCP"

            self.tree.insert(
                "",
                "end",
                values=(proto_type, f"{res['ip']}:{port}", service or "Unknown", test_result),
                tags=(tag,)
            )

        self.root.update()


# ================= RUN =================
